API 설정

키는 Google AI Studio에서 발급. 이 브라우저에만 저장됩니다. 학생과 공유하는 페이지라면 Apps Script 방식을 권장해요.

POST 본문: {"action":"script_coach","system":"...","prompt":"...","json":true} · 응답: {"text":"..."} (또는 result / output / answer 필드)

Apps Script 예시 코드 (웹 앱으로 배포해서 사용) 
// 스크립트 속성에 GEMINI_API_KEY 저장
function doPost(e) {
  const body = JSON.parse(e.postData.contents);
  if (body.action === 'script_coach') {
    const key = PropertiesService.getScriptProperties()
                  .getProperty('GEMINI_API_KEY');
    const url = 'https://generativelanguage.googleapis.com/v1beta/models/'
              + 'gemini-2.5-flash:generateContent?key=' + key;
    const payload = {
      systemInstruction: { parts: [{ text: body.system }] },
      contents: [{ role: 'user', parts: [{ text: body.prompt }] }],
      generationConfig: body.json
        ? { responseMimeType: 'application/json', temperature: 0.6 }
        : { temperature: 0.8 }
    };
    const res = UrlFetchApp.fetch(url, {
      method: 'post', contentType: 'application/json',
      payload: JSON.stringify(payload), muteHttpExceptions: true
    });
    const data = JSON.parse(res.getContentText());
    const text = data.candidates?.[0]?.content?.parts
                   ?.map(p => p.text).join('') || '';
    return ContentService
      .createTextOutput(JSON.stringify({ text }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  // ... 기존 action 분기
}
